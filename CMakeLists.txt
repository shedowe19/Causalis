cmake_minimum_required(VERSION 3.20)
project(CausalisEngine VERSION 0.2.0 LANGUAGES CXX)

# Keep downloadable Windows executables independent of a separately installed
# MSVC CRT. Apply the same runtime to the engine, host, CLI and test targets.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(causalis_engine
  src/engine.cpp src/document.cpp src/script.cpp src/page.cpp
  src/checkpoint.cpp src/vault_policy.cpp)
target_include_directories(causalis_engine PUBLIC include)
target_compile_features(causalis_engine PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(causalis_engine PRIVATE /W4 /utf-8)
else()
  target_compile_options(causalis_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(causalis-cli src/cli.cpp)
target_link_libraries(causalis-cli PRIVATE causalis_engine)
if(MSVC)
  target_compile_options(causalis-cli PRIVATE /W4 /utf-8)
endif()

if(WIN32)
  add_executable(causalis WIN32 windows/main.cpp windows/browser_support.cpp windows/vault_bridge.cpp)
  target_compile_definitions(causalis PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(causalis PRIVATE causalis_engine user32 gdi32 comdlg32 winhttp comctl32 shell32 advapi32)
  if(MSVC)
    target_compile_options(causalis PRIVATE /W4 /utf-8)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  if(WIN32)
    add_test(NAME windows-native-smoke COMMAND causalis --smoke-test)
    set_tests_properties(windows-native-smoke PROPERTIES TIMEOUT 60)
  endif()
  add_executable(causalis-tests tests/engine_tests.cpp)
  target_link_libraries(causalis-tests PRIVATE causalis_engine)
  add_test(NAME engine-tests COMMAND causalis-tests)
  add_executable(causalis-fuzz-smoke tests/fuzz_smoke.cpp)
  target_link_libraries(causalis-fuzz-smoke PRIVATE causalis_engine)
  add_test(NAME fuzz-smoke COMMAND causalis-fuzz-smoke)
  foreach(module document script page vault_policy)
    add_executable(causalis-${module}-tests tests/${module}_tests.cpp)
    target_link_libraries(causalis-${module}-tests PRIVATE causalis_engine)
    add_test(NAME ${module}-tests COMMAND causalis-${module}-tests)
    if(MSVC)
      target_compile_options(causalis-${module}-tests PRIVATE /W4 /utf-8)
    endif()
  endforeach()
  if(MSVC)
    target_compile_options(causalis-tests PRIVATE /W4 /utf-8)
    target_compile_options(causalis-fuzz-smoke PRIVATE /W4 /utf-8)
  endif()
endif()
